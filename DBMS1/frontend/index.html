📖 SQL 语法参考

数据库操作:

CREATE DATABASE db; | DROP DATABASE db; | USE db; | SHOW DATABASES;

表操作 (DDL):

CREATE TABLE t (id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(50) NOT NULL);

DROP TABLE t; | RENAME TABLE old TO new; | DESC t; | SHOW TABLES;

ALTER TABLE t ADD col INT; | ALTER TABLE t DROP col; | ALTER TABLE t MODIFY col VARCHAR(100);

ALTER TABLE t RENAME COLUMN old TO new;

数据操作 (DML):

INSERT INTO t (c1,c2) VALUES (v1,v2);

SELECT * FROM t WHERE c=v ORDER BY c DESC LIMIT 10;

SELECT t1.*,t2.* FROM t1 JOIN t2 ON t1.id=t2.tid; (多表连接)

UPDATE t SET c=v WHERE cond; | DELETE FROM t WHERE cond;

条件支持: =, !=, <>, <, >, <=, >=, LIKE, AND, OR, IS NULL, IS NOT NULL
